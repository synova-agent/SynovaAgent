```bash

# 用法: bash dsh-anchor.sh <包名> <文件相对路径> <符号>
f="$DSH/node_modules/@deepseek-ai/$1/$2"
[ -f "$f" ] && echo "文件 ✓ $(wc -l < "$f") 行" || { echo "文件 ✗ 已不存在"; exit 1; }
n=$(grep -n "$3" "$f" | head -3); [ -n "$n" ] && echo "符号 ✓ $n" || { echo "符号 ✗ 未命中"; exit 1; }
```
